From bits to a class

Each integer from 0 to 15 becomes a row of four bits, most significant bit first. Its target is a 16-element one-hot vector: one entry is 1 and the rest are 0.

For example, 0101 represents 5. The network computes 16 output activations; the index of the largest activation becomes the predicted class.

The Matrix class supplies multiplication, addition, subtraction, transposition and element-wise multiplication using C++ vectors.

Training process

  1. Initialise weights randomly between 0 and 1, with zero biases.
  2. Apply weighted sums and biases, using ReLU in the hidden layer and sigmoid at the output.
  3. Calculate mean-squared error and use the lab's backward update routine to adjust weights and biases.
  4. Repeat over all 16 examples for 3,000 epochs at a learning rate of 0.05, returning the average loss from the final training epoch.

Implementation notes

This preserves the coursework implementation. Its derivative helper applies sigmoid to a value that is already an activation, and the backward pass uses updated weights and a learning-rate-scaled gradient. These are useful points to revisit when comparing the lab with standard backpropagation.

// showcase.cpp
#include <iostream>
#include <vector>
#include "NeuralNetwork.h"

// Binary input representation for numbers 0-15
Matrix inputFromInt(int n) {
    std::vector<std::vector<double>> d(1, std::vector<double>(4));
    d[0][0] = (n >> 3) & 1;
    d[0][1] = (n >> 2) & 1;
    d[0][2] = (n >> 1) & 1;
    d[0][3] = (n >> 0) & 1;
    return Matrix(d);
}

// One-hot encoding target
Matrix targetFromInt(int n) {
    std::vector<std::vector<double>> d(1, std::vector<double>(16, 0.0));
    d[0][n] = 1.0;
    return Matrix(d);
}

int main() {
    // Build dataset
    std::vector<Matrix> inputs, targets;
    for (int i = 0; i < 16; ++i) {
        inputs.push_back(inputFromInt(i));
        targets.push_back(targetFromInt(i));
    }

    // Create user-friendly network
    NeuralNetwork nn(0.05);
    nn.addLayer(4, 16, Activation::RELU);      // hidden
    nn.addLayer(16, 16, Activation::SIGMOID);  // output

    // Train
    double loss = nn.train(inputs, targets, 3000);
    std::cout << "Final loss after training: " << loss << "\n\n";

    // Test in lab screenshot format (Section 3.4 style)
    for (int i = 0; i < 16; ++i) {
        Matrix out = nn.predict(inputFromInt(i));
        auto data = out.getData()[0];
        int pred = 0;
        for (int j = 1; j < 16; ++j)
            if (data[j] > data[pred]) pred = j;

        // Lab-style output (from p.7)
        std::cout << "**********************************************\n";
        std::cout << "Let's test it out!\n";
        std::cout << i << "\n";
        std::cout << ((i >> 3) & 1) << ((i >> 2) & 1)
                  << ((i >> 1) & 1) << ((i >> 0) & 1) << "\n";
        std::cout << "******************** OUTPUT *******************\n";
        for (int j = 0; j < 16; ++j) {
            std::cout << (j + 1) << "---" << data[j] << "\n";
        }
        std::cout << "**********************************************\n";
        std::cout << "Input " << ((i >> 3) & 1) << ((i >> 2) & 1)
                  << ((i >> 1) & 1) << ((i >> 0) & 1)
                  << " -> Prediction: " << pred << "\n\n";
    }

    return 0;
}

#include "Matrix.h"
#include <random>   // For rand() and RAND_MAX
#include <cassert>
#include <cmath>


//Constructor
Matrix::Matrix(int r, int c, bool isRandom): row(r), col(c){
	data.resize(r, std::vector<double>(c, 0.0)); // Initialize to zeros
	 if (isRandom) {
		 // 'static' means these are created only once and reused for efficiency
		static std::random_device rd; // Create a random device to seed the random number generator
		static std::mt19937 gen(rd());  // Create a Mersenne Twister random number generator, seeded with the random device
		static std::uniform_real_distribution<>dis(0.0, 1.0);  // Create a uniform distribution that generates numbers between 0.0 and 1.0
		for(int i=0; i<r; ++i){
			for(int j=0; j< c; ++j){
				data[i][j]=dis(gen); // Assign a random number to the current matrix element
			}	
		}
	}
}

Matrix::Matrix(const std::vector<std::vector<double>>& d)
    : data(d), row(static_cast<int>(d.size())), col(d.empty() ? 0 : static_cast<int>(d[0].size())) {
    // Optional: validate that all rows have the same length
    for (const auto& r : d) {
        if (static_cast<int>(r.size()) != col) {
			throw std::invalid_argument("All rows must have the same number of columns.");	
        }
    }  
}


//getters
int Matrix::getRows() const { return row; }
int Matrix::getCol() const { return col; }
vector<vector<double>> Matrix::getData() const { return data; }



//Print the Matrix
void Matrix::print() const{
	for(const auto& row:data){
		for(double val: row){
			std::cout<<val<<"";
		}
		std::cout<<"\n";
	}

}


//Transpose
Matrix Matrix::transpose() const
{
	Matrix result(col, row);
    for (int i = 0; i < row; ++i) {
        for (int j = 0; j < col; ++j) {
            result.data[j][i] = data[i][j];
        }
    }
    return result;
}


//Matrix Addition
Matrix Matrix::operator+(const Matrix& other) const{
	assert (row==other.row && col==other.col);
	Matrix result(row,col);
	for(int i=0; i<row; ++i){
		for(int j=0; j<col; ++j){
			result.data[i][j]= data[i][j] + other.data[i][j];
		}
	}

	return result;
}


//Matrix Susbtraction
Matrix Matrix::operator-(const Matrix& other) const
{
	assert (row==other.row && col==other.col);
	Matrix result(row,col);
	for(int i=0; i<row; ++i){
		for(int j=0; j<col; ++j){
			result.data[i][j]= data[i][j] - other.data[i][j];
		}
	}

	return result;
}

//Matrix multiplication (A * B)
Matrix Matrix::operator*(const Matrix& other) const
{
	assert(col==other.row); // A.cols == B.rows
	Matrix result(row, other.col);
    for (int i = 0; i < row; ++i) {
        for (int j = 0; j < other.col; ++j) {
            for (int k = 0; k < col; ++k) {
                result.data[i][j] += data[i][k] * other.data[k][j];
            }
        }
    }
    return result;

}

//Scalar multiplication
Matrix Matrix::operator*(double scalar) const
{
	Matrix result(row, col);
	for(int i=0; i<row; ++i){
		for(int j=0; j<col; ++j){
			result.data[i][j]= data[i][j]*scalar;

		}
	}
	return result;
}

// Element-wise multiplication
Matrix Matrix::elementWiseMultiply(const Matrix& other) const
{
	assert(row==other.row && col==other.col);
	Matrix result(row,col);
	for (int i=0; i<row; ++i){
        for (int j = 0; j < col; ++j) {
            result.data[i][j] = data[i][j] * other.data[i][j];
        }
    }
    return result;
	
}

// Static activation functions
double Matrix::sigmoid(double x) {
    return 1.0 / (1.0 + std::exp(-x));
}

double Matrix::dsigmoid(double x) {
    double s = sigmoid(x);
    return s * (1.0 - s);
}
NeuralNetwork.h
// NeuralNetwork.h
#ifndef NEURALNETWORK_H
#define NEURALNETWORK_H

#include "Matrix.h"
#include <vector>
#include <functional>

// Supported activation functions
enum class Activation {
    SIGMOID,
    RELU,
    TANH
};

class NeuralNetwork {
public:
    struct Layer {
        Matrix weights;
        Matrix bias;
        Activation act;
        Layer(int input_size, int output_size, Activation a = Activation::SIGMOID);
    };

private:
    std::vector<Layer> layers;
    double learning_rate;

    // Activation and derivative wrappers
    Matrix activate(const Matrix& z, Activation act) const;
    Matrix dactivate(const Matrix& a, Activation act) const;

public:
    NeuralNetwork(double lr = 0.1);
    void addLayer(int input_size, int output_size, Activation act = Activation::SIGMOID);
    Matrix predict(const Matrix& input) const;
    double train(const std::vector<Matrix>& inputs, const std::vector<Matrix>& targets, int epochs);
    double calculateLoss(const Matrix& output, const Matrix& target) const;
};

// Helper activation functions
double relu(double x);
double drelu(double x);
double tanh_(double x);
double dtanh(double x);

#endif
NeuralNetwork.cpp
// NeuralNetwork.cpp
#include "NeuralNetwork.h"
#include <cassert>
#include <cmath>

// --- Activation functions ---
double relu(double x) { return x > 0 ? x : 0; } // rectified linear unit, it outputs the input directly if it is positive; otherwise, it will output zero
double drelu(double x) { return x > 0 ? 1 : 0; } // derivative of ReLU, it is 1 for positive input and 0 otherwise
double tanh_(double x) { return std::tanh(x); } // hyperbolic tangent function, it outputs values between -1 and 1
double dtanh(double x) {                        // derivative of tanh
    double t = tanh_(x);
    return 1.0 - t * t;
}

// --- Layer constructor ---
// Initializes weights randomly and biases to zero
NeuralNetwork::Layer::Layer(int input_size, int output_size, Activation a)
    : weights(input_size, output_size, true), bias(1, output_size, false), act(a) {}

// --- NeuralNetwork constructor ---
// Initializes learning rate
NeuralNetwork::NeuralNetwork(double lr) : learning_rate(lr) {}

// --- Add a new layer ---
void NeuralNetwork::addLayer(int input_size, int output_size, Activation act) {
    if (!layers.empty()) {
        int last_out = layers.back().weights.getCol();
        if (input_size != last_out) {

            // Auto-correct: use last output as input
            input_size = last_out;
        }
    }
    layers.emplace_back(input_size, output_size, act);
}

// --- Apply activation ---
// Applies the specified activation function element-wise 
Matrix NeuralNetwork::activate(const Matrix& z, Activation act) const {
    auto data = z.getData();
    int rows = z.getRows();
    int cols = z.getCol();
    std::vector<std::vector<double>> out(rows, std::vector<double>(cols));
    for (int i = 0; i < rows; ++i) {
        for (int j = 0; j < cols; ++j) {
            double val = data[i][j];
            switch (act) {
                case Activation::SIGMOID: out[i][j] = Matrix::sigmoid(val); break;
                case Activation::RELU:    out[i][j] = relu(val); break;
                case Activation::TANH:    out[i][j] = tanh_(val); break;
            }
        }
    }
    return Matrix(out);
}

// --- Derivative ---
// Applies the derivative of the specified activation function element-wise
Matrix NeuralNetwork::dactivate(const Matrix& a, Activation act) const {
    auto data = a.getData();
    int rows = a.getRows();
    int cols = a.getCol();
    std::vector<std::vector<double>> out(rows, std::vector<double>(cols));
    for (int i = 0; i < rows; ++i) {
        for (int j = 0; j < cols; ++j) {
            double val = data[i][j];
            switch (act) {
                case Activation::SIGMOID: out[i][j] = Matrix::dsigmoid(val); break;
                case Activation::RELU:    out[i][j] = drelu(val); break;
                case Activation::TANH:    out[i][j] = dtanh(val); break;
            }
        }
    }
    return Matrix(out);
}

// --- Prediction ---
// Feeds input through the network to get output
Matrix NeuralNetwork::predict(const Matrix& input) const {
    Matrix current = input;
    for (const auto& layer : layers) {
        Matrix z = current * layer.weights + layer.bias;
        current = activate(z, layer.act);
    }
    return current;
}

// --- Loss ---
// Mean Squared Error between output and target
double NeuralNetwork::calculateLoss(const Matrix& output, const Matrix& target) const {
    auto o = output.getData();
    auto t = target.getData();
    double loss = 0.0;
    int n = o[0].size();
    for (int i = 0; i < n; ++i) {
        double diff = o[0][i] - t[0][i];
        loss += diff * diff;
    }
    return loss / n;
}

// --- Training ---
double NeuralNetwork::train(const std::vector<Matrix>& inputs,
                            const std::vector<Matrix>& targets,
                            int epochs) {
    double final_loss = 0.0;
    for (int e = 0; e < epochs; ++e) {
        final_loss = 0.0;
        for (size_t i = 0; i < inputs.size(); ++i) {
            // Forward pass
            std::vector<Matrix> activations;
            activations.push_back(inputs[i]);
            Matrix current = inputs[i];
            for (const auto& layer : layers) {
                Matrix z = current * layer.weights + layer.bias;
                current = activate(z, layer.act);
                activations.push_back(current);
            }

            // Loss
            final_loss += calculateLoss(current, targets[i]);

            // Backprop
            Matrix error = current - targets[i];
            for (int l = static_cast<int>(layers.size()) - 1; l >= 0; --l) {
                Matrix deriv = dactivate(activations[l + 1], layers[l].act);
                Matrix gradient = error.elementWiseMultiply(deriv) * learning_rate;

                // Update bias
                layers[l].bias = layers[l].bias - gradient;

                // Delta weights
                Matrix inputT = activations[l].transpose();
                Matrix deltaW = inputT * gradient;
                layers[l].weights = layers[l].weights - deltaW;

                // Propagate error
                if (l > 0) {
                    Matrix weightsT = layers[l].weights.transpose();
                    error = gradient * weightsT;
                }
            }
        }
        final_loss /= inputs.size();
    }
    return final_loss;
}
